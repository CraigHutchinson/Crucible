cmake_minimum_required(VERSION 3.25)
project(Crucible VERSION 0.1.0 LANGUAGES CXX)

if(PROJECT_SOURCE_DIR STREQUAL PROJECT_BINARY_DIR)
    message(FATAL_ERROR "Use an out-of-source build.")
endif()

include(CTest)
option(CRUCIBLE_BUILD_BENCHMARKS "Build headless benchmark tools" OFF)
option(CRUCIBLE_ENABLE_SANITIZERS "Enable AddressSanitizer and UBSan" OFF)
include(cmake/Dependencies.cmake)

add_library(crucible_options INTERFACE)
target_compile_features(crucible_options INTERFACE cxx_std_23)
target_compile_options(crucible_options INTERFACE
    $<$<CXX_COMPILER_ID:MSVC>:/W4;/permissive-;/Zc:preprocessor>
    $<$<NOT:$<CXX_COMPILER_ID:MSVC>>:-Wall;-Wextra;-Wpedantic>)
if(CRUCIBLE_ENABLE_SANITIZERS)
    if(MSVC)
        message(FATAL_ERROR "This sanitizer preset requires GCC or Clang.")
    endif()
    target_compile_options(crucible_options INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(crucible_options INTERFACE -fsanitize=address,undefined)
    foreach(dependency IN ITEMS Sub0Pipeline Sub0Pipeline_Headless)
        target_compile_options(${dependency} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(${dependency} PUBLIC -fsanitize=address,undefined)
    endforeach()
endif()

include(cmake/Workstreams.cmake)
add_subdirectory(src/contracts)
add_subdirectory(src/spatial)
add_subdirectory(src/fields)
add_subdirectory(src/blight)
add_subdirectory(src/swarm)
add_subdirectory(src/interactions)
add_subdirectory(src/scheduling)
add_subdirectory(src/telemetry)
add_subdirectory(src/presentation)

add_library(crucible_core src/simulation.cpp)
add_library(Crucible::Core ALIAS crucible_core)
target_include_directories(crucible_core PUBLIC include)
target_link_libraries(crucible_core PUBLIC crucible_options Crucible::Contracts Sub0ECS::Sub0ECS
    PRIVATE Crucible::Swarm Crucible::Spatial Crucible::Fields Crucible::Blight)
set_target_properties(crucible_core PROPERTIES CXX_EXTENSIONS OFF)
add_subdirectory(src/runtime)

add_executable(crucible src/main.cpp)
target_link_libraries(crucible PRIVATE Crucible::Runtime Crucible::Presentation)
if(BUILD_TESTING)
    add_subdirectory(tests/integration)
    add_subdirectory(tests/workstreams)
endif()
if(CRUCIBLE_BUILD_BENCHMARKS)
    add_subdirectory(benchmarks)
endif()
