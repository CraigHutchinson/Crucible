cmake_minimum_required(VERSION 3.25)
project(Crucible VERSION 0.1.0 LANGUAGES CXX)

if(PROJECT_SOURCE_DIR STREQUAL PROJECT_BINARY_DIR)
    message(FATAL_ERROR "Use an out-of-source build.")
endif()

include(CTest)
option(CRUCIBLE_BUILD_BENCHMARKS "Build headless benchmark tools" OFF)
option(CRUCIBLE_BUILD_RENDER_SPIKE "Build the interim owned-instance rendering experiment" OFF)
option(CRUCIBLE_BUILD_RULE_SPIKE "Build the bounded production-state relay investigation" OFF)
option(CRUCIBLE_BUILD_GPU_RECEIVER "Build the optional Vulkan instancing receiver" OFF)
option(CRUCIBLE_BUILD_DESKTOP "Build the optional SDL interactive prototype" OFF)
option(CRUCIBLE_ENABLE_SANITIZERS "Enable AddressSanitizer and UBSan" OFF)
option(CRUCIBLE_ENABLE_DIAGNOSTICS "Enable startup-bounded runtime outcome diagnostics" OFF)
include(cmake/Dependencies.cmake)
if(CRUCIBLE_BUILD_DESKTOP OR CRUCIBLE_BUILD_GPU_RECEIVER)
    include(cmake/DesktopDependencies.cmake)
endif()

add_library(crucible_options INTERFACE)
target_compile_features(crucible_options INTERFACE cxx_std_23)
target_compile_options(crucible_options INTERFACE
    $<$<CXX_COMPILER_ID:MSVC>:/W4;/permissive-;/Zc:preprocessor>
    $<$<NOT:$<CXX_COMPILER_ID:MSVC>>:-Wall;-Wextra;-Wpedantic>)
if(CRUCIBLE_ENABLE_SANITIZERS)
    if(MSVC)
        message(FATAL_ERROR "This sanitizer preset requires GCC or Clang.")
    endif()
    # Keep Debug assertions/symbols; avoid unoptimized instrumentation dominating replay.
    target_compile_options(crucible_options INTERFACE -O1 -fsanitize=address,undefined -fno-omit-frame-pointer -fno-optimize-sibling-calls)
    target_link_options(crucible_options INTERFACE -fsanitize=address,undefined)
    # Instrument both the graph core and the consumed native worker adapter.
    foreach(dependency IN ITEMS Sub0Pipeline Sub0Pipeline_Priority)
        if(TARGET ${dependency})
            target_compile_options(${dependency} PRIVATE -O1 -fsanitize=address,undefined -fno-omit-frame-pointer -fno-optimize-sibling-calls)
            target_link_options(${dependency} PUBLIC -fsanitize=address,undefined)
        endif()
    endforeach()
endif()

include(cmake/Workstreams.cmake)
add_subdirectory(src/contracts)
add_subdirectory(src/spatial)
add_subdirectory(src/fields)
add_subdirectory(src/blight)
add_subdirectory(src/swarm)
add_subdirectory(src/interactions)
add_subdirectory(src/scheduling)
add_subdirectory(src/telemetry)
add_subdirectory(src/presentation)

add_library(crucible_core src/simulation.cpp)
add_library(Crucible::Core ALIAS crucible_core)
target_include_directories(crucible_core PUBLIC include)
target_link_libraries(crucible_core PUBLIC crucible_options Crucible::Contracts Sub0ECS::Sub0ECS
    PRIVATE Crucible::Swarm Crucible::Spatial Crucible::Fields Crucible::Blight Crucible::Interactions Crucible::Scheduling)
set_target_properties(crucible_core PROPERTIES CXX_EXTENSIONS OFF)
add_subdirectory(src/runtime)
add_subdirectory(src/application)

if(CRUCIBLE_BUILD_GPU_RECEIVER)
    include(cmake/GpuShaders.cmake)
endif()
add_subdirectory(src/presentation/gpu)

add_executable(crucible src/main.cpp)
target_link_libraries(crucible PRIVATE Crucible::Runtime Crucible::Presentation Crucible::Application)
if(CRUCIBLE_BUILD_DESKTOP OR CRUCIBLE_BUILD_GPU_RECEIVER)
    add_subdirectory(src/presentation/desktop)
endif()
if(CRUCIBLE_BUILD_DESKTOP)
    add_subdirectory(src/desktop)
endif()
if(BUILD_TESTING)
    find_package(Python3 3.10 REQUIRED COMPONENTS Interpreter)
    add_test(NAME workflow_permissions COMMAND ${Python3_EXECUTABLE}
        ${PROJECT_SOURCE_DIR}/tests/workflow/test_permissions.py)
    set_tests_properties(workflow_permissions PROPERTIES LABELS workflow)
    add_test(NAME workflow_prerequisites COMMAND ${Python3_EXECUTABLE}
        ${PROJECT_SOURCE_DIR}/tests/workflow/test_prerequisites.py)
    set_tests_properties(workflow_prerequisites PROPERTIES LABELS workflow)
    add_subdirectory(tests/integration)
    add_subdirectory(tests/workstreams)
    add_subdirectory(tests/application)
endif()
if(CRUCIBLE_BUILD_BENCHMARKS)
    add_subdirectory(benchmarks)
endif()

if(CRUCIBLE_BUILD_RULE_SPIKE)
    add_subdirectory(spikes/structural)
endif()

if(CRUCIBLE_BUILD_RENDER_SPIKE)
    if(NOT CRUCIBLE_BUILD_DESKTOP)
        message(FATAL_ERROR "The rendering spike needs CRUCIBLE_BUILD_DESKTOP=ON.")
    endif()
    add_subdirectory(spikes/rendering)
endif()

if(CRUCIBLE_BUILD_GPU_RECEIVER)
    add_executable(crucible_gpu_receiver src/desktop/GpuReceiver.cpp)
    target_link_libraries(crucible_gpu_receiver PRIVATE Crucible::GpuOffscreen Crucible::Runtime Crucible::ScenePainter SDL3::SDL3)
    add_dependencies(crucible_gpu_receiver crucible_gpu_shaders)
